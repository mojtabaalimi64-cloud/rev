async function getIranStations(coords, radius) { let revStations = []; try { revStations = await getRevStations(coords, radius); } catch (error) { console.warn("REV API unavailable:", error); } // جست‌وجوی آنلاین حتی اگر REV نتیجه داشته باشد انجام می‌شود. let osmStations = []; try { status( `از دیتابیس REV تعداد ${revStations.length} ایستگاه دریافت شد؛ ` + "در حال بررسی ایستگاه‌های آنلاین…" ); osmStations = await getOsmStations(coords, radius); } catch (error) { console.warn("OSM unavailable:", error); } // ترکیب نتایج و حذف ایستگاه‌های تکراری const combined = [...revStations]; for (const station of osmStations) { const duplicate = combined.some(existing => haversine( [existing.lat, existing.lon], [station.lat, station.lon] ) < 0.15 ); if (!duplicate) combined.push(station); } return combined .map(station => ({ ...station, routeDistance: distanceToRoute( [station.lat, station.lon], coords ) })) .filter(station => station.routeDistance <= radius) .sort((a, b) => a.routeDistance - b.routeDistance); }
